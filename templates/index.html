async function sendAIMessage() { var input = document.getElementById('aiUserInput'); var body = document.getElementById('aiChatBody'); var text = input.value.trim(); if (!text) return; // إضافة رسالة المستخدم body.innerHTML += `
${text}
`; input.value = ''; // إنشاء صندوق رد البوت الفارغ let botMsg = document.createElement('div'); botMsg.className = 'ai-msg bot'; botMsg.style.cssText = "margin: 10px 0; text-align: right; background: #f1f1f1; color: #333; padding: 8px 12px; border-radius: 8px; width: fit-content; max-width: 80%; white-space: pre-wrap;"; body.appendChild(botMsg); body.scrollTop = body.scrollHeight; try { const response = await fetch('/ask-ai', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: text }) }); const reader = response.body.getReader(); const decoder = new TextDecoder(); while (true) { const { done, value } = await reader.read(); if (done) break; botMsg.innerText += decoder.decode(value, { stream: true }); body.scrollTop = body.scrollHeight; } } catch (error) { botMsg.innerText += " عذراً، حدث خطأ في الاتصال."; } }
